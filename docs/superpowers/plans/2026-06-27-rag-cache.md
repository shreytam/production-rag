# Multi-Tier RAG Cache Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a production-grade multi-tier cache (L1 exact response, L2 semantic response, L3 query embedding, L4 retrieval) behind one swappable `Cache` interface, on a shared Redis 8 instance, with tenant-safe isolation and version-based invalidation.

**Architecture:** A `Cache` Protocol with a `RedisCache` (Redis 8 strings + native Vector Sets) and a `NullCache` no-op. Response tiers (L1/L2) are checked in `RAGPipeline.answer()`; embedding (L3) and retrieval (L4) are transparent decorators on the `Embedder`/`Retriever`. Config-gated, off on the eval path, traced in Langfuse, degrades to no-op when Redis is unreachable.

**Tech Stack:** Python 3.12, pydantic v2, redis-py (Redis 8 Vector Sets via `VADD`/`VSIM`/`VREM`), fakeredis (tests), pytest, uv.

## Global Constraints

- Tenant isolation is **hostile by design**: `tenant_id` is hashed exactly into every key AND the L2 semantic index is partitioned per tenant — a cross-tenant hit must be structurally impossible, never a post-filter.
- Only the **normalized query text** is compared semantically. `corpus_version`, `prompt_version`, pipeline version, and `gen_model` are exact-match key dimensions, never merged.
- Default semantic threshold = **cosine similarity 0.95** (`cache_semantic_threshold`), tunable.
- Shared Redis stays `--maxmemory-policy noeviction` (Langfuse requires it). The cache self-bounds via **per-key TTL** + per-tenant `cache_semantic_max_entries`; never assume LRU eviction.
- **Never cache a refused or guardrail-blocked answer.** Input guardrails run before the cache lookup.
- Cache is **off on the eval path** (`build(enable_cache=False)` in `eval/run_eval.py` and `eval/ragas_adapter.py`).
- Any cache/Redis failure must degrade to no-op — a query never breaks because of the cache.

```python
# core/cache.py
"""Cache keys, versioning, and Answer<->record (de)serialization.

Key dimensions that must match EXACTLY for a hit (never merged): tenant_id,
corpus_version, prompt_version, pipeline version, gen_model. Only the normalized
query text is compared semantically (in the L2 KNN)."""

from __future__ import annotations

import hashlib
from pathlib import Path

from core.config import Settings
from core.types import Answer

# Bump when SYSTEM_PROMPT / prompt templates change (invalidates cached answers).
PROMPT_VERSION = "1"

_CORPUS_VERSION_FILE = Path(".cache") / "corpus_version"


def normalize(q: str) -> str:
    """Cheap query-quality control: trim, collapse whitespace, lowercase."""
    return " ".join(q.split()).lower()


def _sha(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def read_corpus_version() -> str:
    try:
        return _CORPUS_VERSION_FILE.read_text("utf-8").strip() or "0"
    except FileNotFoundError:
        return "0"


def bump_corpus_version() -> str:
    cur = int(read_corpus_version())
    new = str(cur + 1)
    _CORPUS_VERSION_FILE.parent.mkdir(parents=True, exist_ok=True)
    _CORPUS_VERSION_FILE.write_text(new, "utf-8")
    return new


class CacheKeys:
    """Builds namespaced keys for one pipeline (version + model). corpus_version
    is read fresh per access so a re-ingest invalidates a running process."""

    def __init__(self, settings: Settings, version: str) -> None:
        self._version = version
        self._gen_model = settings.gen_model
        self._embed_model = settings.embed_model

    @property
    def partition(self) -> str:
        return f"{read_corpus_version()}.{PROMPT_VERSION}:{self._version}:{self._gen_model}"

    def resp_key(self, tenant: str, q: str) -> str:
        return f"ragcache:resp:{self.partition}:{tenant}:{_sha(normalize(q))}"

    def retr_key(self, tenant: str, q: str) -> str:
        return f"ragcache:retr:{self.partition}:{tenant}:{_sha(normalize(q))}"

    def emb_key(self, q: str) -> str:
        return f"ragcache:emb:{self._embed_model}:{_sha(normalize(q))}"

    def sem_ns(self, tenant: str) -> str:
        return f"ragcache:sem:{self.partition}:{tenant}"


def answer_to_record(ans: Answer, source_query: str) -> dict:
    """Serialize an Answer for caching. Strips context embeddings (large, unneeded
    for replay) so a cached response round-trips to an identical Answer."""
    data = ans.model_dump()
    for sc in data.get("contexts", []):
        chunk = sc.get("chunk")
        if isinstance(chunk, dict):
            chunk["embedding"] = None
    data["source_query"] = source_query
    return data


def record_to_answer(record: dict) -> Answer:
    data = {k: v for k, v in record.items() if k != "source_query"}
    return Answer.model_validate(data)
```

```python
# tests/_fakes.py (append)
import math


class FakeCache:
    """In-memory Cache impl for offline tests. Semantic index is per (tenant,
    partition) namespace; knn is brute-force cosine — mirrors RedisCache semantics
    without Redis."""

    def __init__(self) -> None:
        self._kv: dict[str, dict] = {}
        self._vec: dict[str, list[float]] = {}
        self._sem: dict[str, list[tuple[list[float], dict]]] = {}

    def get(self, key: str):
        return self._kv.get(key)

    def set(self, key: str, value: dict, ttl: int) -> None:
        self._kv[key] = value

    def get_vector(self, key: str):
        return self._vec.get(key)

    def set_vector(self, key: str, vec: list[float], ttl: int) -> None:
        self._vec[key] = vec

    @staticmethod
    def _cos(a: list[float], b: list[float]) -> float:
        dot = sum(x * y for x, y in zip(a, b))
        na = math.sqrt(sum(x * x for x in a)) or 1.0
        nb = math.sqrt(sum(y * y for y in b)) or 1.0
        return dot / (na * nb)

    def knn(self, tenant, partition, query_vec, top_k=1):
        ns = f"{partition}:{tenant}"
        scored = [(rec, self._cos(query_vec, v)) for v, rec in self._sem.get(ns, [])]
        scored.sort(key=lambda t: t[1], reverse=True)
        return scored[:top_k]

    def add_semantic(self, tenant, partition, query_vec, record, ttl, max_entries) -> None:
        ns = f"{partition}:{tenant}"
        bucket = self._sem.setdefault(ns, [])
        bucket.append((query_vec, record))
        if len(bucket) > max_entries:
            bucket.pop(0)

    def clear(self, prefix: str = "ragcache:") -> int:
        n = len(self._kv) + len(self._vec) + len(self._sem)
        self._kv.clear(); self._vec.clear(); self._sem.clear()
        return n
```

```python
# providers/cache/redis_cache.py
"""Redis 8 cache. String tiers (response / embedding / retrieval) use plain
key/value with per-key TTL. The semantic tier uses Redis 8 native Vector Sets
(VADD/VSIM/VREM), so no extra module/image is required.

The shared Redis is `noeviction` (Langfuse), so we never rely on LRU: every
entry has a TTL and the per-tenant semantic set is capped explicitly. All Redis
errors are swallowed — a cache failure must never break a query."""

from __future__ import annotations

import json
from typing import Any

from core.config import Settings


def _connect(url: str):
    import redis  # local import so the dep is optional

    return redis.from_url(url, decode_responses=True)


class RedisCache:
    def __init__(self, settings: Settings) -> None:
        self._r = _connect(settings.cache_redis_url)
        # Fail fast so the registry can fall back to NullCache on a bad connection.
        self._r.ping()

    # --- string tiers --------------------------------------------------------
    def get(self, key: str) -> dict | None:
        try:
            raw = self._r.get(key)
            return json.loads(raw) if raw is not None else None
        except Exception:
            return None

    def set(self, key: str, value: dict, ttl: int) -> None:
        try:
            self._r.set(key, json.dumps(value), ex=ttl)
        except Exception:
            pass

    def get_vector(self, key: str) -> list[float] | None:
        rec = self.get(key)
        return rec.get("v") if rec else None

    def set_vector(self, key: str, vec: list[float], ttl: int) -> None:
        self.set(key, {"v": list(vec)}, ttl)

    # --- semantic tier (Redis 8 Vector Sets) ---------------------------------
    def knn(self, tenant: str, partition: str, query_vec: list[float], top_k: int = 1):
        ns = f"ragcache:sem:{partition}:{tenant}"
        try:
            dim = len(query_vec)
            flat = [str(x) for x in query_vec]
            raw = self._r.execute_command(
                "VSIM", ns, "VALUES", dim, *flat, "WITHSCORES", "COUNT", top_k
            )
        except Exception:
            return []
        out: list[tuple[dict, float]] = []
        # VSIM WITHSCORES returns a flat [element, score, element, score, ...]
        for i in range(0, len(raw), 2):
            element, score = raw[i], float(raw[i + 1])
            rec = self.get(element)            # record stored at the element key
            if rec is None:                    # record TTL'd out -> stale vector
                try:
                    self._r.execute_command("VREM", ns, element)
                except Exception:
                    pass
                continue
            out.append((rec, score))
        return out

    def add_semantic(self, tenant, partition, query_vec, record, ttl, max_entries) -> None:
        ns = f"ragcache:sem:{partition}:{tenant}"
        element = record["__key__"]            # the resp_key, set by the caller
        try:
            dim = len(query_vec)
            flat = [str(x) for x in query_vec]
            self._r.execute_command("VADD", ns, "VALUES", dim, *flat, element)
            self._r.expire(ns, ttl)            # whole-set TTL backstop
            card = int(self._r.execute_command("VCARD", ns))
            if card > max_entries:             # approximate cap (noeviction safety)
                victims = self._r.execute_command("VRANDMEMBER", ns, card - max_entries)
                for v in victims if isinstance(victims, list) else [victims]:
                    self._r.execute_command("VREM", ns, v)
        except Exception:
            pass

    # --- maintenance ---------------------------------------------------------
    def clear(self, prefix: str = "ragcache:") -> int:
        n = 0
        try:
            for key in self._r.scan_iter(match=f"{prefix}*"):
                self._r.delete(key)
                n += 1
        except Exception:
            pass
        return n
```

```python
# providers/embedders/caching.py
"""L3: transparent query-embedding cache. Wraps any Embedder and memoizes
embed_query by normalized-text key. embed_documents is passed through (bulk
ingest, not worth caching)."""

from __future__ import annotations

from core.cache import CacheKeys
from core.interfaces import Cache, Embedder


class CachingEmbedder:
    def __init__(self, inner: Embedder, cache: Cache, keys: CacheKeys, ttl: int) -> None:
        self._inner = inner
        self._cache = cache
        self._keys = keys
        self._ttl = ttl

    @property
    def dimension(self) -> int:
        return self._inner.dimension

    def embed_documents(self, texts: list[str]) -> list[list[float]]:
        return self._inner.embed_documents(texts)

    def embed_query(self, text: str) -> list[float]:
        key = self._keys.emb_key(text)
        cached = self._cache.get_vector(key)
        if cached is not None:
            return cached
        vec = self._inner.embed_query(text)
        self._cache.set_vector(key, vec, self._ttl)
        return vec
```

```python
# retrieval/caching.py
"""L4: transparent retrieval-result cache. Wraps any Retriever and caches the
ScoredChunk list per (tenant, normalized query). Embeddings are stripped before
caching and chunks reconstructed on a hit."""

from __future__ import annotations

from core.cache import CacheKeys
from core.interfaces import Cache, Retriever
from core.types import Chunk, Query, RetrievalSource, ScoredChunk


class CachingRetriever:
    def __init__(self, inner: Retriever, cache: Cache, keys: CacheKeys, ttl: int) -> None:
        self._inner = inner
        self._cache = cache
        self._keys = keys
        self._ttl = ttl

    def retrieve(self, query: Query) -> list[ScoredChunk]:
        key = self._keys.retr_key(query.acl.tenant_id, query.text)
        cached = self._cache.get(key)
        if cached is not None:
            return [self._deserialize(d) for d in cached["scored"]]
        scored = self._inner.retrieve(query)
        self._cache.set(key, {"scored": [self._serialize(s) for s in scored]}, self._ttl)
        return scored

    @staticmethod
    def _serialize(sc: ScoredChunk) -> dict:
        d = sc.model_dump()
        if isinstance(d.get("chunk"), dict):
            d["chunk"]["embedding"] = None
        return d

    @staticmethod
    def _deserialize(d: dict) -> ScoredChunk:
        return ScoredChunk.model_validate(d)
```

```python
# scripts/tune_cache_threshold.py
"""Offline helper: sweep the semantic-cache threshold over a labeled set of
(query_a, query_b, should_match) triples and print precision / false-positive
rate per threshold, so the production threshold is chosen on data, not a guess.

Usage: uv run python scripts/tune_cache_threshold.py pairs.json
  pairs.json = [{"a": "...", "b": "...", "match": true}, ...]
"""

from __future__ import annotations

import json
import sys

from core.registry import build_embedder


def _cos(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    na = sum(x * x for x in a) ** 0.5 or 1.0
    nb = sum(y * y for y in b) ** 0.5 or 1.0
    return dot / (na * nb)


def main() -> None:
    pairs = json.loads(open(sys.argv[1]).read())
    emb = build_embedder()
    sims = [(_cos(emb.embed_query(p["a"]), emb.embed_query(p["b"])), bool(p["match"])) for p in pairs]
    print(f"{'thresh':>7} {'hits':>5} {'TP':>4} {'FP':>4} {'precision':>9} {'fp_rate':>8}")
    for t in [round(0.80 + 0.01 * i, 2) for i in range(21)]:
        hits = [(s, m) for s, m in sims if s >= t]
        tp = sum(1 for s, m in hits if m)
        fp = sum(1 for s, m in hits if not m)
        prec = tp / len(hits) if hits else 0.0
```
